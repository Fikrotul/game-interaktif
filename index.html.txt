        /* ================= 2. WEB AUDIO SYNTHESIZER ================= */
        const AudioContext = window.AudioContext || window.webkitAudioContext;
        let audioCtx = null;

        function initAudio() {
            if (!audioCtx) {
                audioCtx = new AudioContext();
            }
            if (audioCtx.state === 'suspended') {
                audioCtx.resume();
            }
        }

        function playSound(type) {
            if (!isSoundOn) return;
            try {
                initAudio();
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.connect(gain);
                gain.connect(audioCtx.destination);

                const now = audioCtx.currentTime;

                if (type === 'flip') {
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(400, now);
                    osc.frequency.exponentialRampToValueAtTime(800, now + 0.1);
                    gain.gain.setValueAtTime(0.15, now);
                    gain.gain.linearRampToValueAtTime(0.01, now + 0.1);
                    osc.start(now);
                    osc.stop(now + 0.1);
                } else if (type === 'correct') {
                    osc.type = 'triangle';
                    // Arpeggio C Major
                    osc.frequency.setValueAtTime(523.25, now); // C5
                    osc.frequency.setValueAtTime(659.25, now + 0.08); // E5
                    osc.frequency.setValueAtTime(783.99, now + 0.16); // G5
                    osc.frequency.setValueAtTime(1046.50, now + 0.24); // C6
                    gain.gain.setValueAtTime(0.2, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.4);
                    osc.start(now);
                    osc.stop(now + 0.4);
                } else if (type === 'wrong') {
                    osc.type = 'sawtooth';
                    osc.frequency.setValueAtTime(220, now);
                    osc.frequency.setValueAtTime(180, now + 0.12);
                    gain.gain.setValueAtTime(0.2, now);
                    gain.gain.linearRampToValueAtTime(0.01, now + 0.3);
                    osc.start(now);
                    osc.stop(now + 0.3);
                } else if (type === 'victory') {
                    // Fanfare jingle
                    const notes = [523.25, 659.25, 783.99, 1046.50];
                    notes.forEach((freq, idx) => {
                        const noteOsc = audioCtx.createOscillator();
                        const noteGain = audioCtx.createGain();
                        noteOsc.connect(noteGain);
                        noteGain.connect(audioCtx.destination);
                        
                        noteOsc.type = 'triangle';
                        noteOsc.frequency.value = freq;
                        const startTime = now + (idx * 0.12);
                        noteGain.gain.setValueAtTime(0.2, startTime);
                        noteGain.gain.exponentialRampToValueAtTime(0.001, startTime + 0.25);
                        
                        noteOsc.start(startTime);
                        noteOsc.stop(startTime + 0.25);
                    });
                }
            } catch (e) {
                console.warn("Web Audio API Audio error: ", e);
            }
        }

        function loadQuestionsFromStorage() {
            const stored = localStorage.getItem('sd_quiz_cards_data');
            if (stored) {
                try {
                    questionsData = JSON.parse(stored);
                } catch (e) {
                    questionsData = [...DEFAULT_QUESTIONS];
                }
            } else {
                questionsData = [...DEFAULT_QUESTIONS];
                saveQuestionsToStorage();
            }
        }

        function saveQuestionsToStorage() {
            localStorage.setItem('sd_quiz_cards_data', JSON.stringify(questionsData));
        }

        function startGame(category) {
            currentCategory = category;
            currentGameQuestions = questionsData.filter(q => q.category === category);

            if (currentGameQuestions.length === 0) {
                alert(`Belum ada soal untuk mata pelajaran ${category}. Silakan tambah soal di Mode Guru!`);
                return;
            }

            // Shuffle questions
            currentGameQuestions = currentGameQuestions.sort(() => Math.random() - 0.5);

            score = 0;
            correctCount = 0;
            wrongCount = 0;
            answeredCount = 0;

            document.getElementById('gameCategoryTitle').innerText = category;
            document.getElementById('gameCategoryBadge').innerText = `Mata Pelajaran ${category}`;
            updateHeaderStats();
            renderCardGrid();
            showScreen('game');
        }

        function restartGame() {
            if (currentCategory) {
                startGame(currentCategory);
            } else {
                showScreen('home');
            }
        }

        function selectAnswer(selectedIndex, correctIndex, buttonEl) {
            const q = currentGameQuestions[currentCardIndex];
            const allOptionBtns = document.getElementById('modalOptionsContainer').children;
            
            // Disable all buttons to prevent double tap
            for (let btn of allOptionBtns) {
                btn.disabled = true;
                btn.classList.add('cursor-not-allowed', 'opacity-80');
            }

            const feedbackEl = document.getElementById('modalFeedback');
            feedbackEl.classList.remove('hidden');

            const isCorrect = selectedIndex === correctIndex;
            const cardEl = document.getElementById(`card-${currentCardIndex}`);
            const statusIcon = cardEl.querySelector('.status-icon');
            const statusText = cardEl.querySelector('.status-text');

            if (isCorrect) {
                playSound('correct');
                score += 100;
                correctCount++;
                buttonEl.classList.remove('bg-gray-50', 'border-gray-200');
                buttonEl.classList.add('bg-emerald-100', 'border-emerald-400', 'text-emerald-800');
                
                feedbackEl.innerText = '🎉 Jawabanmu Benar! (+100 Poin)';
                feedbackEl.className = 'text-center py-3 px-4 rounded-2xl font-bold text-sm bg-emerald-100 text-emerald-700 border border-emerald-300 animate-bounce';

                statusIcon.innerText = '✅';
                statusText.innerText = 'Benar (+100)';
                statusText.className = 'text-xs font-bold text-emerald-600 font-display status-text';
            } else {
                playSound('wrong');
                wrongCount++;
                buttonEl.classList.remove('bg-gray-50', 'border-gray-200');
                buttonEl.classList.add('bg-rose-100', 'border-rose-400', 'text-rose-800');
                
                // Highlight the correct one
                allOptionBtns[correctIndex].classList.remove('bg-gray-50', 'border-gray-200');
                allOptionBtns[correctIndex].classList.add('bg-emerald-100', 'border-emerald-300', 'text-emerald-800');

                feedbackEl.innerText = `❌ Yah, Kurang Tepat! Jawaban benar: ${q.options[correctIndex]}`;
                feedbackEl.className = 'text-center py-3 px-4 rounded-2xl font-bold text-sm bg-rose-100 text-rose-700 border border-rose-300';

                statusIcon.innerText = '❌';
                statusText.innerText = 'Salah (0)';
                statusText.className = 'text-xs font-bold text-rose-500 font-display status-text';
            }

            answeredCount++;
            updateHeaderStats();

            // Flip the card on the grid
            cardEl.classList.add('flipped');

            // Close Modal after delay
            setTimeout(() => {
                closeQuestionModal();
                if (answeredCount === currentGameQuestions.length) {
                    setTimeout(showFinalResultScreen, 500);
                }
            }, 1800);
        }

        function handleFormSubmit(e) {
            e.preventDefault();
            const editIndex = parseInt(document.getElementById('editIndex').value);
            const category = document.getElementById('formCategory').value;
            const question = document.getElementById('formQuestion').value.trim();
            const opt0 = document.getElementById('opt0').value.trim();
            const opt1 = document.getElementById('opt1').value.trim();
            const opt2 = document.getElementById('opt2').value.trim();
            const correctRadio = parseInt(document.querySelector('input[name="correctRadio"]:checked').value);

            const newQ = {
                category,
                question,
                options: [opt0, opt1, opt2],
                answer: correctRadio

                alert('Pertanyaan baru berhasil ditambahkan!');

            saveQuestionsToStorage();
            resetQuestionForm();
            renderTeacherQuestionsList();
        }

        /* ================= 7. CELEBRATION CONFETTI ANIMATION ================= */
        let confettiParticles = [];
        let animationFrameId = null;

        function initConfetti() {
            const canvas = document.getElementById('confettiCanvas');
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
            window.addEventListener('resize', () => {
                canvas.width = window.innerWidth;
                canvas.height = window.innerHeight;
            });
        }

        function startConfetti() {
            const canvas = document.getElementById('confettiCanvas');
            const ctx = canvas.getContext('2d');
            confettiParticles = [];

            const colors = ['#A0C4FF', '#FFADAD', '#FDFFB6', '#CAFFBF', '#E0C3FC', '#FFD6A5'];

            for (let i = 0; i < 120; i++) {
                confettiParticles.push({
                    x: Math.random() * canvas.width,
                    y: Math.random() * canvas.height - canvas.height,
                    size: Math.random() * 10 + 5,
                    color: colors[Math.floor(Math.random() * colors.length)],
                    speedY: Math.random() * 3 + 2,
                    speedX: Math.random() * 2 - 1,
                    rotation: Math.random() * 360,
                    rotationSpeed: Math.random() * 10 - 5
                });
            }

            if (animationFrameId) cancelAnimationFrame(animationFrameId);

            let startTime = Date.now();
            function render() {
                ctx.clearRect(0, 0, canvas.width, canvas.height);

                confettiParticles.forEach(p => {
                    p.y += p.speedY;
                    p.x += p.speedX;
                    p.rotation += p.rotationSpeed;

                    ctx.save();
                    ctx.translate(p.x, p.y);
                    ctx.rotate((p.rotation * Math.PI) / 180);
                    ctx.fillStyle = p.color;
                    ctx.fillRect(-p.size / 2, -p.size / 2, p.size, p.size);
                    ctx.restore();
                });

                if (Date.now() - startTime < 4000) {
                    animationFrameId = requestAnimationFrame(render);
                } else {
                    ctx.clearRect(0, 0, canvas.width, canvas.height);
                }
            }
            render();
        }
